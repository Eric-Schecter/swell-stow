func material(color: Color) -> StandardMaterial3D:
 var m := StandardMaterial3D.new()
 m.albedo_color = color
 m.roughness = 0.78
 return m

func block_mesh(size: Vector3) -> ArrayMesh:
 var st := SurfaceTool.new()
 st.begin(Mesh.PRIMITIVE_TRIANGLES)
 st.set_smooth_group(-1)
 var b: float = min(0.045, min(size.x, min(size.y,size.z))*0.19)
 var rings: Array = []
 for data in [[-size.y/2,b],[ -size.y/2+b,0.0],[size.y/2-b,0.0],[size.y/2,b]]:
  var x: float = size.x/2-data[1]
  var z: float = size.z/2-data[1]
  var c: float = b*0.65
  rings.append([Vector3(-x+c,data[0],-z),Vector3(x-c,data[0],-z),Vector3(x,data[0],-z+c),Vector3(x,data[0],z-c),Vector3(x-c,data[0],z),Vector3(-x+c,data[0],z),Vector3(-x,data[0],z-c),Vector3(-x,data[0],-z+c)])
 for r in range(3):
  for i in range(8):
   var j := (i+1)%8
   for v in [rings[r][i],rings[r+1][j],rings[r+1][i],rings[r][i],rings[r][j],rings[r+1][j]]:
    st.add_vertex(v)
 for i in range(1,7):
  for v in [rings[0][0],rings[0][i+1],rings[0][i],rings[3][0],rings[3][i],rings[3][i+1]]:
   st.add_vertex(v)
 st.generate_normals()
 return st.commit()

func box(size: Vector3, pos: Vector3, color: Color, parent: Node = self) -> MeshInstance3D:
 var mesh := MeshInstance3D.new()
 mesh.mesh = block_mesh(size)
 mesh.material_override = material(color)
 parent.add_child(mesh)
 mesh.position = pos
 return mesh

func cylinder(radius: float, height: float, pos: Vector3, color: Color, parent: Node=self) -> MeshInstance3D:
 var n := MeshInstance3D.new()
 var mesh := CylinderMesh.new()
 mesh.top_radius=radius
 mesh.bottom_radius=radius
 mesh.height=height
 mesh.radial_segments=24
 n.mesh=mesh
 n.material_override=material(color)
 parent.add_child(n)
 n.position=pos
 return n

func make_boat(color: Color, tug: bool = false) -> Node3D:
 var ship := Node3D.new()
 add_child(ship)
 var hull := Node3D.new()
 ship.add_child(hull)
 box(Vector3(0.72,0.23,1.18),Vector3(0,0.0,-0.06),color,hull)
 # A faceted pointed bow faces +Z.
 var bow := MeshInstance3D.new()
 var vertices := PackedVector3Array([Vector3(-0.36,-0.115,0.5),Vector3(0.36,-0.115,0.5),Vector3(0,-0.115,0.95),Vector3(-0.36,0.115,0.5),Vector3(0.36,0.115,0.5),Vector3(0,0.115,0.95)])
 var st := SurfaceTool.new()
 st.begin(Mesh.PRIMITIVE_TRIANGLES)
 st.set_smooth_group(-1)
 for i in [3,4,5,0,2,1,0,1,4,0,4,3,1,2,5,1,5,4,2,0,3,2,3,5]: st.add_vertex(vertices[i])
 st.generate_normals()
 bow.mesh=st.commit()
 var bm := material(color)
 bm.cull_mode=BaseMaterial3D.CULL_DISABLED
 bow.material_override=bm
 hull.add_child(bow)
 box(Vector3(0.61,0.045,1.05),Vector3(0,0.14,-0.05),Color("f0dfbf"),hull)
 box(Vector3(0.48,0.4,0.47),Vector3(0,0.36,-0.2),Color("fff0d1"),hull)
 box(Vector3(0.54,0.07,0.54),Vector3(0,0.6,-0.2),color,hull)
 box(Vector3(0.35,0.14,0.02),Vector3(0,0.4,0.045),Color("829eaa"),hull)
 for x in [-0.25,0.25]:
  box(Vector3(0.015,0.14,0.22),Vector3(x,0.39,-0.2),Color("8da9b1"),hull)
 cylinder(0.04,0.38,Vector3(0.15,0.81,-0.25),Color("b1a48b"),hull)
 for x in [-0.37,0.37]:
  for z in [-0.35,0.1]:
   cylinder(0.065,0.11,Vector3(x,0.02,z),Color("6c8587"),hull).rotation_degrees.z=90

 if tug:
  cylinder(0.09,0.36,Vector3(0,0.8,-0.2),Color("697e80"),hull)
  cylinder(0.12,0.07,Vector3(0,1.0,-0.2),Color("e4b88c"),hull)
 return ship

func new_member(pos: Vector2, color: Color, tug: bool=false) -> Dictionary:
 var node := make_boat(color,tug)
 node.rotation.y=PI
 node.position=Vector3(pos.x,-0.16,pos.y)
 return {"node":node,"x":pos.x,"v":0.0,"z":pos.y,"bias":0.0,"age":0.0,"heading":PI,"bank_cd":0.0,"tension":0.0}

 for b in fleet: b.node.queue_free()
 for b in loose: b.node.queue_free()
 if not incoming.is_empty(): incoming.node.queue_free()
 for w in wakes: w.node.queue_free()
 fleet.clear()
 loose.clear()
 wakes.clear()
 incoming.clear()
 clock_time=0.0

func begin() -> void:
 if mode!="ready" and mode!="over": return
 if mode=="over": reset_run()
 mode="sailing"
 headline.text="对齐船尾光圈 · 点击接船"
 subline.text="接偏会累积；下一艘在收缆时就会进场"
 action.hide()
 spawn_incoming()

func tail_target() -> Vector2:
 var last: Dictionary=fleet.back()
 return Vector2(last.x,last.z+SPACING)

func spawn_incoming() -> void:
 boat_serial+=1
 var side: int=-1 if boat_serial%2==1 else 1
 var target := tail_target()
 var node := make_boat(COLORS[rescued%COLORS.size()])
 var speed: float=2.65+minf(float(rescued)*0.105,2.8)
 incoming={"node":node,"x":float(side)*5.2,"z":target.y,"speed":-side*speed,"side":side,"age":0.0,"id":boat_serial}
 node.rotation.y=PI/2 if side<0 else -PI/2
 node.position=Vector3(incoming.x,-0.16,incoming.z)
 marker.show()

func activate() -> void:
 if paused: return
 if mode in ["ready","over"]: begin()
 elif mode=="sailing": catch_boat()

func catch_boat() -> void:
 if mode!="sailing" or paused or incoming.is_empty(): return
 var target := tail_target()
 var error: float=incoming.x-target.x
 var old_id: int=incoming.id
 attempts+=1
 max_action_gap=maxf(max_action_gap,run_time-last_catch)
 last_catch=run_time
 if absf(error)>CATCH_RADIUS:
  miss("抛缆落空  ·  船队被扯了一下",error)
  return
 var perfect: bool=absf(error)<=PERFECT_RADIUS
 var member: Dictionary={"node":incoming.node,"x":incoming.x,"v":incoming.speed*0.16,"z":incoming.z,"bias":error,"age":0.0,"heading":PI/2 if incoming.side<0 else -PI/2,"bank_cd":0.0,"tension":0.0}
 # The catch error becomes this joint's rest offset. It is not reset or recentered.
 fleet.append(member)
 incoming={}
 rescued+=1
 if perfect:
  chain+=1
  integrity=minf(100.0,integrity+4.0)
  feedback.text="精准接船 ✦ %d  ·  缆绳恢复" % chain
  if chain%3==0:
   for i in range(1,fleet.size()):
    fleet[i].bias*=0.65
    fleet[i].v*=0.65
   integrity=minf(100.0,integrity+8.0)
   feedback.text="三连精准  ·  收紧全队，纠正偏移"
 else:
  chain=0
  integrity-=absf(error)*5.0
  feedback.text="接偏 %+.2f  ·  下一艘接缆点也向%s偏" % [error,"右" if error>0 else "左"]
  # Immediate reaction reaches the previous boat while the new one is turning.
  fleet[fleet.size()-2].v+=member.v*0.30
 flash_timer=0.4
 if fleet.size()>MAX_VISIBLE:
  deliver_front()
 spawn_incoming()
 assert(incoming.id!=old_id)
 update_ui()

func miss(message: String, offset: float) -> void:
 chain=0
 integrity-=23.0
 if not incoming.is_empty():
  loose.append({"node":incoming.node,"x":incoming.x,"z":incoming.z,"vx":incoming.speed,"age":0.0})
 incoming={}
 fleet.back().v+=clampf(offset,-1,1)*0.50
 feedback.text=message
 flash_timer=0.5
 if integrity<=0:
  finish()
 else:
  spawn_incoming()
 update_ui()

func deliver_front() -> void:
 # A supply launch takes the foremost cargo; preserve the surviving tail's offset.
 var delivered_boat: Dictionary=fleet[1]
 fleet.remove_at(1)
 loose.append({"node":delivered_boat.node,"x":delivered_boat.x,"z":delivered_boat.z,"vx":-2.4,"age":0.0})
 if fleet.size()>1: fleet[1].bias=fleet[1].x-fleet[0].x
 delivered+=1
 integrity=minf(100,integrity+5)
 banner="前船已送达  ·  +5 缆绳强度"
 banner_timer=1.1

func step_world(dt: float) -> void:
 if mode!="sailing" or paused: return
 clock_time+=dt
 run_time+=dt
 tide=sin(run_time*0.73)*(.12+minf(rescued*.012,.24))
 center=sin(run_time*.15)*minf(float(rescued)*.045,.55)
 # Two-way lateral spring chain. A new boat moves its predecessor too.
 var forces: Array[float]=[]
 for i in range(fleet.size()): forces.append(tide*(0.75+float(i)*0.18))
 forces[0]+=(center-fleet[0].x)*5.5-fleet[0].v*3.6
 for i in range(1,fleet.size()):
  var a: Dictionary=fleet[i-1]
  var b: Dictionary=fleet[i]
  var stretch: float=b.x-a.x-b.bias
  var force: float=-stretch*13.0-(b.v-a.v)*3.7
  forces[i]+=force
  forces[i-1]-=force*0.72
  b.tension=absf(force)+absf(b.x-a.x)*1.8
  if b.tension>4.6: integrity-=(b.tension-4.6)*dt*0.40
 for i in range(fleet.size()):
  var b: Dictionary=fleet[i]
  b.age+=dt
  b.bank_cd=maxf(0,b.bank_cd-dt)
  b.v+=(forces[i]-b.v*0.95)*dt
  b.x+=b.v*dt
  b.z=lerpf(b.z,-4.3+float(i)*SPACING,1-exp(-dt*4.5))
  if absf(b.x-center)>2.48:
   if b.bank_cd<=0:
    integrity-=14
    b.bank_cd=0.7
    feedback.text="船队擦到浮标线  ·  把下一艘接向航道内侧"
    flash_timer=0.5
   b.v-=signf(b.x-center)*dt*2.0
 if not incoming.is_empty():
  incoming.age+=dt
  incoming.x+=(incoming.speed+tide*0.30)*dt
  incoming.z=lerpf(incoming.z,tail_target().y,1-exp(-dt*7.0))
  if incoming.x*float(incoming.side)< -5.4:
   attempts+=1
   miss("错过一艘  ·  下一艘已进场",-float(incoming.side))
 if integrity<=0: finish()

func demo_input() -> void:
 if mode!="sailing" or incoming.is_empty(): return
 var error: float=incoming.x-tail_target().x
 var wanted: float=0.0
 if rescued in [1,2]: wanted=0.38
 elif rescued==4: wanted=-0.30
 if (incoming.speed>0 and error>=wanted) or (incoming.speed<0 and error<=wanted): catch_boat()

func _physics_process(dt: float) -> void:
 if tests or demo: return
 step_world(dt)

func update_visuals(dt: float) -> void:
 for i in range(fleet.size()):
  var b: Dictionary=fleet[i]
  b.heading=lerp_angle(b.heading,PI-atan(b.v*.55),1-exp(-dt*6.5))
  b.node.position=Vector3(b.x,-0.16+sin(clock_time*2.5+i*.6)*.018,b.z)
  b.node.rotation=Vector3(0,b.heading,-clampf(b.v*.1,-.1,.1))
 if not incoming.is_empty():
  incoming.node.position=Vector3(incoming.x,-0.16,incoming.z)
 var target := tail_target()
 marker.position=Vector3(target.x,-0.43,target.y)
 marker_ring.scale=Vector3.ONE*(0.98+sin(clock_time*6)*0.025)
 if not incoming.is_empty():
  var aligned: bool=absf(incoming.x-target.x)<PERFECT_RADIUS
  marker_ring.material_override.albedo_color=Color("e8ffb7") if aligned else Color("f4e6b2")
 var mesh := ImmediateMesh.new()
 if fleet.size()>1:
  mesh.surface_begin(Mesh.PRIMITIVE_TRIANGLES)
  for i in range(1,fleet.size()):
   var a: Dictionary=fleet[i-1]
   var b: Dictionary=fleet[i]
   var from := Vector3(a.x,0.05,a.z+.56)
   var to := Vector3(b.x,0.05,b.z-.58)
   var side := (to-from).normalized().cross(Vector3.UP)*.022
   for v in [from-side,from+side,to+side,from-side,to+side,to-side]: mesh.surface_add_vertex(v)
  mesh.surface_end()
 ropes.mesh=mesh
 lane_left.position.x=center-2.95
 lane_right.position.x=center+2.95
 var height: float=clampf(17.0+maxi(0,fleet.size()-4)*1.67,17.0,22.0)
 cam.size=lerpf(cam.size,height,1-exp(-dt*2.0))
 var focus := Vector3(0,0,(-4.3+target.y)*.5-.35)
 var desired := focus+Vector3(4.0,17,11.5)
 cam.position=cam.position.lerp(desired,1-exp(-dt*2.2))
 cam.look_at(cam.position-Vector3(4.0,17,11.5))

func _process(dt: float) -> void:
 if tests: return
 if demo and not paused and run_time<demo_duration:
  for i in range(4):
   step_world(1.0/120)
   demo_input()
 if not paused:
  if mode=="sailing":
   for n in scenery:
    n.position.z+=dt*2.0
    if n.position.z>18: n.position.z-=29
   wake_timer+=dt
   if wake_timer>.17:
    wake_timer=0.0
    for b in fleet: add_wake(b)
  flash_timer=maxf(0,flash_timer-dt)
  banner_timer=maxf(0,banner_timer-dt)
  update_effects(dt)
 update_visuals(dt if not paused else 0.0)
 update_ui()
 if demo and not captured:
  if rescued>=6 and run_time>12:
   captured=true
   save_capture("res://captures/convoy-chain.png")
  if mode=="over":
   print("DEMO ended unexpectedly: rescued=",rescued," integrity=",integrity)
   get_tree().quit(1)
 if demo and run_time>=demo_duration and not capture_pending:
  print("DEMO PASS: rescued=",rescued," delivered=",delivered," integrity=",snappedf(integrity,.1)," max_input_gap=",snappedf(max_action_gap,.01),"s")
  get_tree().quit()

func add_wake(b: Dictionary) -> void:
 var w := box(Vector3(0.52,0.012,0.045),Vector3(b.x,-0.44,b.z+.75),Color("cde4d9"),effects_root)
 w.cast_shadow=GeometryInstance3D.SHADOW_CASTING_SETTING_OFF
 wakes.append({"node":w,"age":0.0})

func update_effects(dt: float) -> void:
 for i in range(wakes.size()-1,-1,-1):
  var w: Dictionary=wakes[i]
  w.age+=dt
  w.node.position.z+=dt*2.0
  w.node.scale.x=1+w.age*.65
  w.node.transparency=minf(1,w.age/1.25)
  if w.age>1.25:
   w.node.queue_free()
   wakes.remove_at(i)
 for i in range(loose.size()-1,-1,-1):
  var b: Dictionary=loose[i]
  b.age+=dt
  b.x+=b.vx*dt
  b.z+=dt*.7
  b.node.position=Vector3(b.x,-.16,b.z)
  b.node.rotation.y=lerp_angle(b.node.rotation.y,signf(b.vx)*PI/2,dt*2)
  if b.age>2.0:
   b.node.queue_free()
   loose.remove_at(i)

func finish() -> void:
 if mode=="over": return
 mode="over"
 integrity=0
 marker.hide()
 headline.text="缆绳撑不住了"
 subline.text="船队的偏移和拉扯会累积；下次可以故意向内侧接船纠偏"
 feedback.text="本次接船 %d 艘  ·  已送达 %d 艘" % [rescued,delivered]
 action.text="重新启航"
 action.show()
 if rescued>best:
  best=rescued
  if save_enabled:
   var f := FileAccess.open("user://convoy-best.txt",FileAccess.WRITE)
   if f: f.store_string(str(best))

func toggle_pause() -> void:
 if mode!="sailing": return
 paused=not paused
 pause_button.text="▶" if paused else "Ⅱ"

func _unhandled_input(event: InputEvent) -> void:
 if event is InputEventKey and event.pressed and not event.echo:
  if event.keycode==KEY_ESCAPE: toggle_pause()

  if mode=="sailing": catch_boat()
