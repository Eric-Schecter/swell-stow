func material(color: Color) -> StandardMaterial3D:
 var m := StandardMaterial3D.new()
 m.albedo_color = color
 m.roughness = 0.78
 return m

func block_mesh(size: Vector3) -> ArrayMesh:
 var st := SurfaceTool.new()
 st.begin(Mesh.PRIMITIVE_TRIANGLES)
 st.set_smooth_group(-1)
 var b: float = min(0.045, min(size.x, min(size.y,size.z))*0.19)
 var rings: Array = []
 for data in [[-size.y/2,b],[ -size.y/2+b,0.0],[size.y/2-b,0.0],[size.y/2,b]]:
  var x: float = size.x/2-data[1]
  var z: float = size.z/2-data[1]
  var c: float = b*0.65
  rings.append([Vector3(-x+c,data[0],-z),Vector3(x-c,data[0],-z),Vector3(x,data[0],-z+c),Vector3(x,data[0],z-c),Vector3(x-c,data[0],z),Vector3(-x+c,data[0],z),Vector3(-x,data[0],z-c),Vector3(-x,data[0],-z+c)])
 for r in range(3):
  for i in range(8):
   var j := (i+1)%8
   for v in [rings[r][i],rings[r+1][j],rings[r+1][i],rings[r][i],rings[r][j],rings[r+1][j]]:
    st.add_vertex(v)
 for i in range(1,7):
  for v in [rings[0][0],rings[0][i+1],rings[0][i],rings[3][0],rings[3][i],rings[3][i+1]]:
   st.add_vertex(v)
 st.generate_normals()
 return st.commit()

# Same fixed-step simulation drives play and the deterministic checks.
func integrate_mooring(pos: Vector2, vel: Vector2, length: float, flow: Vector2, dt: float) -> Array:
 length=maxf(1.1,length-0.95*dt)
 var radius := pos-ANCHOR
 var distance := radius.length()
 var direction := radius/maxf(distance,0.001)
 var radial_speed := vel.dot(direction)
 var tension := maxf(0.0,11.0*(distance-length)+3.0*radial_speed)
 var acceleration := -direction*tension-1.25*vel+flow*0.55
 vel+=acceleration*dt
 pos+=vel*dt
 return [pos,vel,length]

func collision(pos: Vector2) -> bool:
 # Boat radius against the solid quay and the two projecting jetties.
 if pos.x>1.88 and absf(pos.y)<4.15: return true
 if pos.x>0.52 and pos.x<2.5:
  if absf(pos.y+1.65)<0.43 or absf(pos.y-2.0)<0.43: return true
 return false

func docked(pos: Vector2, vel: Vector2) -> bool:
 return absf(pos.x-TARGET.x)<0.32 and absf(pos.y-TARGET.y)<0.24 and vel.length()<0.28

func _physics_process(dt: float) -> void:
 if paused or test_mode: return
 if state=="approach":
  velocity=Vector2(current.x+sin(clock_time*0.8)*0.018,speed+current.y)
  p+=velocity*dt
  if p.y>3.8:
   finish(false,"错过泊位","没有及时抛缆，船已驶过码头。")
  if not demo_mode.is_empty():
   var trigger := -1.0
   if demo_mode=="early": trigger=-3.1
   elif demo_mode=="late": trigger=1.5
   elif demo_mode=="crash": trigger=2.4
   if p.y>=trigger: throw_rope()
 elif state=="mooring":
  elapsed+=dt
  var values := integrate_mooring(p,velocity,rope_length,current,dt)
  p=values[0]
  velocity=values[1]
  rope_length=values[2]
  if collision(p):
   finish(false,"碰到码头了","船还有向前的惯性，下次提前抛缆。")
  elif elapsed>5.2:
   if docked(p,velocity):
    var perfect := p.distance_to(TARGET)<0.13
    finish(true,"完美靠泊" if perfect else "靠泊成功","船已停稳。下一艘船的航速和海流会变化。")
   elif p.y<TARGET.y-0.24:
    finish(false,"抛缆太早","船停在泊位前方，下次稍晚一点抛缆。")
   else:
    finish(false,"抛缆太晚","船滑过了理想停靠点，下次提前一点抛缆。")
 update_ship(dt)
 if state in ["approach","mooring"]:
  wake_time+=dt
  if wake_time>0.13 and velocity.length()>0.15:
   wake_time=0.0
   add_wake()

func update_ship(dt: float) -> void:
 ship.position=Vector3(p.x,-0.16,p.y)
 if velocity.length()>0.08:
  var wanted := atan2(velocity.x,velocity.y)
  # Rudder restores fore-aft alignment as the boat loses speed.
  if state=="mooring": wanted=lerp_angle(wanted,0.0,clampf(elapsed/5.0,0,1))
  heading=lerp_angle(heading,wanted,1.0-exp(-dt*3.0))
 ship.rotation.y=heading
 hull.rotation.z=sin(clock_time*1.6)*0.025 if state!="result" else 0.0
 if rope.visible:
  var from := ship.position+Vector3(0,0.18,0.38).rotated(Vector3.UP,heading)
  var to := Vector3(ANCHOR.x,0.65,ANCHOR.y)
  var mesh := ImmediateMesh.new()
  mesh.surface_begin(Mesh.PRIMITIVE_LINE_STRIP)
  for i in range(17):
   var t := float(i)/16
   var v := from.lerp(to,t)
   v.y-=sin(t*PI)*0.09
   mesh.surface_add_vertex(v)
  mesh.surface_end()
  rope.mesh=mesh

func _unhandled_input(event: InputEvent) -> void:
 if event is InputEventKey and event.pressed and not event.echo:
  if event.keycode==KEY_ESCAPE: toggle_pause()
  elif event.keycode in [KEY_SPACE,KEY_ENTER]: activate()
 elif event is InputEventMouseButton and event.pressed and event.button_index==MOUSE_BUTTON_LEFT:
  if state=="approach": throw_rope()
 elif event is InputEventScreenTouch and event.pressed:
  if state=="approach": throw_rope()

func _process(dt: float) -> void:
 if not paused:
  clock_time+=dt
  for i in range(wakes.size()-1,-1,-1):
   var w: Dictionary=wakes[i]
   w.age+=dt
   w.node.scale.x=1.0+w.age*0.8
   w.node.transparency=minf(1,w.age/2.0)
   if w.age>2.0:
    w.node.queue_free()
    wakes.remove_at(i)
 if not demo_mode.is_empty() and not captured:
  if state=="mooring" and elapsed>1.6 and demo_mode=="turn":
   captured=true
   capture("turn")
  elif state=="result":
   finish_timer+=dt
   if finish_timer>0.4:
    captured=true
    capture(demo_mode)

func simulate_throw(z: float, flow: Vector2=Vector2.ZERO, incoming: float=1.55, start_x: float=-1.5) -> Dictionary:
 var pos := Vector2(start_x,z)
 var vel := Vector2(flow.x,incoming+flow.y)
 var length := pos.distance_to(ANCHOR)
 for i in range(625):
  var values := integrate_mooring(pos,vel,length,flow,1.0/120)
  pos=values[0]
  vel=values[1]
  length=values[2]
  if collision(pos): return {"hit":true,"success":false,"position":pos}
 return {"hit":false,"success":docked(pos,vel),"position":pos}

func run_tests() -> void:
 var good := simulate_throw(-1.0)
 var early := simulate_throw(-3.1)
 var late := simulate_throw(1.5)
 print("SIMULATION: good=",good," early=",early," late=",late)
 assert(good.success,"On-time throw must berth using real integration")
 assert(not early.success,"Early throw must fail")
 assert(not late.success,"Late throw must fail")
 assert(simulate_throw(2.4).hit,"Very late throw must physically intersect the jetty")
 assert(collision(Vector2(2.0,0)))
