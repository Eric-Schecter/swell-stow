func material(color: Color) -> StandardMaterial3D:
 var m := StandardMaterial3D.new()
 m.albedo_color = color
 m.roughness = 0.78
 return m

func block_mesh(size: Vector3) -> ArrayMesh:
 var st := SurfaceTool.new()
 st.begin(Mesh.PRIMITIVE_TRIANGLES)
 st.set_smooth_group(-1)
 var b: float = min(0.045, min(size.x, min(size.y,size.z))*0.19)
 var rings: Array = []
 for data in [[-size.y/2,b],[ -size.y/2+b,0.0],[size.y/2-b,0.0],[size.y/2,b]]:
  var x: float = size.x/2-data[1]
  var z: float = size.z/2-data[1]
  var c: float = b*0.65
  rings.append([Vector3(-x+c,data[0],-z),Vector3(x-c,data[0],-z),Vector3(x,data[0],-z+c),Vector3(x,data[0],z-c),Vector3(x-c,data[0],z),Vector3(-x+c,data[0],z),Vector3(-x,data[0],z-c),Vector3(-x,data[0],-z+c)])
 for r in range(3):
  for i in range(8):
   var j := (i+1)%8
   for v in [rings[r][i],rings[r+1][j],rings[r+1][i],rings[r][i],rings[r][j],rings[r+1][j]]:
    st.add_vertex(v)
 for i in range(1,7):
  for v in [rings[0][0],rings[0][i+1],rings[0][i],rings[3][0],rings[3][i],rings[3][i+1]]:
   st.add_vertex(v)
 st.generate_normals()
 return st.commit()

func box(size: Vector3, pos: Vector3, color: Color, parent: Node = self) -> MeshInstance3D:
 var mesh := MeshInstance3D.new()
 mesh.mesh = block_mesh(size)
 mesh.material_override = material(color)
 parent.add_child(mesh)
 mesh.position = pos
 return mesh

func reset_tower() -> void:
 for l in layers: l.queue_free()
 layers.clear()
 if is_instance_valid(current): current.queue_free()
 for s in scraps: s.node.queue_free()
 scraps.clear()
 for d in decorations: d.queue_free()
 decorations.clear()
 width=Vector2(START_WIDTH,START_WIDTH)
 top_pos=Vector3.ZERO
 score=0
 streak=0
 axis=0
 settling=false
 queued=false
 ended=false
 paused=false
 layers.append(box(Vector3(width.x,HEIGHT,width.y),top_pos,COLORS[0]))
 update_ui()

func start() -> void:
 reset_tower()
 playing=true
 menu.hide()
 feedback.text=""
 hint.text="点击 / 空格落下 · Esc 暂停"
 spawn()

func spawn() -> void:
 axis=score%2
 phase=-PI/2
 current=box(Vector3(width.x,HEIGHT,width.y),top_pos+Vector3(0,HEIGHT+0.16,0),COLORS[(score+1)%COLORS.size()])
 update_moving()

func update_moving() -> void:
 if not is_instance_valid(current): return
 current.position=top_pos+Vector3(0,HEIGHT+0.16,0)
 current.position[axis*2] += sin(phase)*3.45

func drop() -> void:
 if not playing or paused: return
 if settling:
  queued=true
  return
 var direction := axis*2
 var offset: float = current.position[direction]-top_pos[direction]
 var full: float = width[axis]
 var overlap: float = full-abs(offset)
 if overlap <= 0.0:
  scraps.append({"node":current,"velocity":Vector3(sign(offset)*0.5,0,0),"spin":Vector3(0.5,0.3,0.8),"life":0.0})
  current=null
  finish()
  return
 var perfect: bool = abs(offset)<0.115
 var pos := top_pos+Vector3(0,HEIGHT,0)
 var old_width := width
 if perfect:
  streak+=1
  current.position[direction]=top_pos[direction]
  feedback.text="完美对齐  ✦  " + str(streak)
 else:
  streak=0
  width[axis]=overlap
  pos[direction]+=offset*0.5
  var cut_size := Vector3(old_width.x,HEIGHT,old_width.y)
  cut_size[direction]=abs(offset)
  var cut_pos := current.position
  cut_pos.y=pos.y
  cut_pos[direction]=top_pos[direction]+sign(offset)*(full/2+abs(offset)/2)
  var chip := box(cut_size,cut_pos,COLORS[(score+1)%COLORS.size()])
  var vel := Vector3(0,0.1,0)
  vel[direction]=sign(offset)*0.9
  scraps.append({"node":chip,"velocity":vel,"spin":Vector3(0.6,0.2,0.8)*sign(offset),"life":0.0})
  current.mesh=block_mesh(Vector3(width.x,HEIGHT,width.y))
  feedback.text="再稳一点，下一层会更窄"
 settling=true
 var placed := current
 layers.append(placed)
 current=null
 top_pos=pos
 score+=1
 if score>best:
  best=score
  if save_enabled:
   var f := FileAccess.open("user://harborlight-player.save",FileAccess.WRITE)
   if f: f.store_string(str(best))
 var tween := create_tween()
 tween.tween_property(placed,"position",pos,0.14).set_trans(Tween.TRANS_QUAD).set_ease(Tween.EASE_IN)
 tween.tween_callback(func():
  if streak>0 and streak%3==0 or score%8==0:
   width.x=min(START_WIDTH,width.x+0.3)
   width.y=min(START_WIDTH,width.y+0.3)
   top_pos.y+=HEIGHT
   layers.append(box(Vector3(width.x,HEIGHT,width.y),top_pos,Color("f4dfb5")))
   feedback.text="加固平台  ·  宽度恢复"
  if score%8==0:
   add_lantern()
   feedback.text="航标点亮  ·  船只正在通过"
  settling=false
  if playing:
   spawn()
   if queued:
    queued=false
    drop()
 )
 update_ui()

func toggle_pause() -> void:
 if not playing: return
 paused=not paused
 pause_button.text="▶" if paused else "Ⅱ"
 hint.text="已暂停 · 点击右上角或按 Esc 继续" if paused else "点击 / 空格落下 · Esc 暂停"
 # Placement tweens also pause, without disabling the UI.
 for tw in get_tree().get_processed_tweens():
  if paused: tw.pause()
  else: tw.play()

func update_ui() -> void:
 score_label.text=str(score)
 best_label.text="最佳高度  %d   /   下个航标  %d 层" % [best,(score/8+1)*8]
 if not playing and not ended: hint.text="每三次完美对齐获得加宽平台 · 每八层点亮航标"

func _unhandled_input(event: InputEvent) -> void:
 if event is InputEventKey and event.pressed and not event.echo:
  if event.keycode==KEY_ESCAPE: toggle_pause()
  elif event.keycode in [KEY_SPACE,KEY_ENTER]:
   if playing: drop()
   else: start()
 elif event is InputEventMouseButton and event.pressed and event.button_index==MOUSE_BUTTON_LEFT:
  if playing: drop()
 elif event is InputEventScreenTouch and event.pressed:
  if playing: drop()

func _process(delta: float) -> void:
 time+=delta
 boat.position.x=2.9+sin(time*0.1)*0.6
 boat.position.z=-2.8+cos(time*0.15)*0.7
 boat.position.y=-0.66+sin(time*1.2)*0.035
 boat.rotation.z=sin(time)*0.025
 if playing and not paused:
  if not settling:
   phase+=delta*(1.05+min(score,40)*0.018) * (3.0 if demo else 1.0)
   update_moving()
  for i in range(scraps.size()-1,-1,-1):
   var s: Dictionary = scraps[i]
   s.life+=delta
   s.velocity.y-=delta*7.0
   s.node.position+=s.velocity*delta
   s.node.rotation+=s.spin*delta
   if s.node.position.y < -0.85 or s.life>5:
    splash(s.node.position)
    s.node.queue_free()
    scraps.remove_at(i)
 # Missed blocks continue falling after the round ends.
 elif not paused:
  for i in range(scraps.size()-1,-1,-1):
   var s: Dictionary=scraps[i]
   s.velocity.y-=delta*7
   s.node.position+=s.velocity*delta
   s.node.rotation+=s.spin*delta
   if s.node.position.y < -0.85:
    splash(s.node.position)
    s.node.queue_free()
    scraps.remove_at(i)
 for i in range(ripples.size()-1,-1,-1):
  var r: Dictionary=ripples[i]
  r.age+=delta
  r.node.scale=Vector3.ONE*(0.15+r.age*0.9)
  r.node.transparency=min(1.0,r.age/1.2)
  if r.age>1.2:
   r.node.queue_free()
   ripples.remove_at(i)
 var target := Vector3(top_pos.x*0.25,max(0.6,top_pos.y-0.55),top_pos.z*0.25)
 var desired := target+Vector3(7,6.4,9)
 cam.position=cam.position.lerp(desired,1-exp(-delta*2.8))
 cam.look_at(cam.position-Vector3(7,6.4,9))
 if demo and not demo_done:
  demo_timer+=delta
  if not playing and demo_timer>0.3: start()
  if playing and not settling and is_instance_valid(current):
   var offset: float=abs(current.position[axis*2]-top_pos[axis*2])
   var threshold := 0.11 if score%5!=2 else 0.45
   if offset<threshold: drop()
  if score>=12:
   demo_done=true
   await get_tree().create_timer(0.4).timeout
   RenderingServer.force_draw()
   await RenderingServer.frame_post_draw
   get_viewport().get_texture().get_image().save_png("res://captures/harborlight-gameplay.png")
   print("CAPTURE PASS: 12 layers, platform recovery, navigation lantern")
   get_tree().quit()
